/* Generated at build time. Only this project's versioned caches are read. */
'use strict';
const OFFLINE = /*__OFFLINE_MANIFEST__*/;
const ORIGIN = self.location.origin;
const ENTRY_BY_PATH = new Map(OFFLINE.files.map(file=>[file.url,file]));
const absolute = path => new URL(path,ORIGIN).href;
const toHex = value => [...new Uint8Array(value)].map(n=>n.toString(16).padStart(2,'0')).join('');

async function validateResponse(response,entry){
 if(response.status!==200||response.type==='opaque')throw new Error('Offline download did not return a complete response');
 const buffer=await response.arrayBuffer();
 if(buffer.byteLength!==entry.bytes||toHex(await crypto.subtle.digest('SHA-256',buffer))!==entry.sha256)throw new Error('Offline resource failed integrity check');
 // fetch() exposes a decoded body; never retain transfer encoding or compressed size.
 const headers=new Headers(response.headers);headers.delete('Content-Encoding');headers.delete('Transfer-Encoding');headers.set('Content-Length',String(buffer.byteLength));
 return new Response(buffer,{status:200,headers});
}

async function installShell(){
 const cache=await caches.open(OFFLINE.shellCache);
 const entries=OFFLINE.files.filter(file=>file.group==='shell');let next=0;
 async function worker(){while(next<entries.length){const entry=entries[next++];
  const url=absolute(entry.url),existing=await cache.match(url);
  if(existing){try{await validateResponse(existing.clone(),entry);continue;}catch{/* Repair only this incomplete version's entry. */}}
  const response=await fetch(new Request(url,{cache:'reload',credentials:'same-origin'}));
  await cache.put(url,await validateResponse(response,entry));
 }}
 await Promise.all(Array.from({length:Math.min(4,entries.length)},worker));
}
self.addEventListener('install',event=>{event.waitUntil(installShell());});
// No install-time takeover, cache deletion or save manipulation.
let explicitUpdate=false;
self.addEventListener('activate',event=>{if(explicitUpdate)event.waitUntil(self.clients.claim());});
// Only the explicit, version-checked update page may request activation.
async function otherGamePages(){
 const windows=await self.clients.matchAll({type:'window',includeUncontrolled:true});
 return windows.filter(client=>{const url=new URL(client.url);return url.origin===ORIGIN&&navigationPath(url.pathname)!==null;}).length;
}
async function completePackage(){
 const media=await caches.open(OFFLINE.mediaCache),marker=await media.match(absolute(OFFLINE.completeMarker));
 if(!marker)return false;try{const value=await marker.json();if(value.version!==OFFLINE.version||value.complete!==true)return false;}catch{return false;}
 for(const entry of OFFLINE.files)if(!await(await caches.open(entry.group==='shell'?OFFLINE.shellCache:OFFLINE.mediaCache)).match(absolute(entry.url)))return false;
 return true;
}
self.addEventListener('message',event=>{
 const port=event.ports?.[0];
 if(event.data?.type==='AEQUILIBRIUM_MANIFEST'){port?.postMessage({manifest:OFFLINE});return;}
 if(!['AEQUILIBRIUM_UPDATE_STATUS','AEQUILIBRIUM_ACTIVATE'].includes(event.data?.type)||!port)return;
 event.waitUntil((async()=>{
  try{
   const source=event.source?.url?new URL(event.source.url):null;
   if(!source||source.origin!==ORIGIN||![OFFLINE.base+'update/',OFFLINE.base+'update/index.html'].includes(source.pathname))throw new Error('更新请求必须来自本游戏的更新页。');
   if(event.data.version!==OFFLINE.version)throw new Error('新版已变化，请重新检查后再更新。');
   const openGames=await otherGamePages();
   if(event.data.type==='AEQUILIBRIUM_UPDATE_STATUS'){port.postMessage({version:OFFLINE.version,openGames});return;}
   if(openGames){port.postMessage({version:OFFLINE.version,openGames,blocked:true});return;}
   if(!await completePackage())throw new Error('新版资源尚未准备完整，当前版本仍保留。');
   explicitUpdate=true;await self.skipWaiting();port.postMessage({version:OFFLINE.version,accepted:true});
  }catch(error){port.postMessage({error:error.message??'更新未能完成。'});}
 })());
});

function navigationPath(path){
 if(path===OFFLINE.base||path===OFFLINE.base+'index.html')return OFFLINE.base+'index.html';
 if(path===OFFLINE.base+'playtest'||path===OFFLINE.base+'playtest/'||path===OFFLINE.base+'playtest/index.html')return OFFLINE.base+'playtest/index.html';
 return null;
}
function parseRange(value,size){
 const match=/^bytes=(\d*)-(\d*)$/.exec(value?.trim()??'');
 if(!match||(!match[1]&&!match[2])||size===0)return null;
 let start,end;
 if(!match[1]){const count=Number(match[2]);if(!Number.isSafeInteger(count)||count<=0)return null;start=Math.max(0,size-count);end=size-1;}
 else{start=Number(match[1]);end=match[2]?Number(match[2]):size-1;if(!Number.isSafeInteger(start)||!Number.isSafeInteger(end)||start>=size||end<start)return null;end=Math.min(size-1,end);}
 return{start,end};
}
async function rangedResponse(response,request){
 const body=await response.arrayBuffer(),range=parseRange(request.headers.get('Range'),body.byteLength);
 const headers=new Headers(response.headers);headers.delete('Content-Encoding');headers.delete('Transfer-Encoding');headers.set('Accept-Ranges','bytes');
 if(!range){headers.set('Content-Range',`bytes */${body.byteLength}`);headers.set('Content-Length','0');return new Response(null,{status:416,headers});}
 const {start,end}=range;headers.set('Content-Range',`bytes ${start}-${end}/${body.byteLength}`);headers.set('Content-Length',String(end-start+1));
 return new Response(request.method==='HEAD'?null:body.slice(start,end+1),{status:206,headers});
}
function headOnly(response){return new Response(null,{status:response.status,statusText:response.statusText,headers:response.headers});}
async function cached(entry){
 const cache=await caches.open(entry.group==='shell'?OFFLINE.shellCache:OFFLINE.mediaCache);
 const response=await cache.match(absolute(entry.url));return response?.status===200?response:undefined;
}
async function respond(request,url){
 // Update discovery and the recovery entry always reach the network. Old workers
 // also forward these previously unknown paths, so already-started saves can upgrade.
 if(url.pathname===OFFLINE.base+'release.json'||url.pathname.startsWith(OFFLINE.base+'update/'))return fetch(new Request(request.url,{method:request.method,headers:request.headers,cache:'no-store',credentials:'same-origin'}));
 // Always report this controlling worker's manifest, not a newer incompatible shell.
 if(url.pathname===OFFLINE.base+'offline-manifest.json')return new Response(request.method==='HEAD'?null:JSON.stringify(OFFLINE),{status:200,headers:{'Content-Type':'application/json; charset=utf-8','Cache-Control':'no-store'}});
 // Explicit integrity repair must reach the network rather than re-read a
 // damaged cached response. A failed repair keeps the prior cache intact.
 if(request.cache==='reload')return fetch(request);
 const nav=request.mode==='navigate'?navigationPath(url.pathname):null;
 const entry=ENTRY_BY_PATH.get(nav??url.pathname);
 if(entry){
  const response=await cached(entry);
  if(response){if(request.headers.has('Range'))return rangedResponse(response,request);return request.method==='HEAD'?headOnly(response):response;}
 }
 try{return await fetch(request);}catch{
  // Missing media/scripts must not receive the app HTML masquerading as success.
  return new Response(request.method==='HEAD'?null:'This resource is not available offline.',{status:503,headers:{'Content-Type':'text/plain; charset=utf-8','Cache-Control':'no-store'}});
 }
}
self.addEventListener('fetch',event=>{
 const request=event.request;if(request.method!=='GET'&&request.method!=='HEAD')return;
 const url=new URL(request.url);if(url.origin!==ORIGIN||!url.pathname.startsWith(OFFLINE.base))return;
 event.respondWith(respond(request,url));
});
